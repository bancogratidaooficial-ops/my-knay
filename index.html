function iniciarGrav(){ chunks=[]; recorder=new MediaRecorder(stream,{mimeType:'video/mp4'}); recorder.ondataavailable=e=>{ if(e.data.size>0) chunks.push(e.data) }; recorder.onstop=()=>{ let blob=new Blob(chunks,{type:'video/mp4'}); let url=URL.createObjectURL(blob); let nome='KNAY_'+new Date().toISOString().slice(0,19).replace(/:/g,'-')+'.mp4'; // 1. Salva na lista do app videos.unshift({nome:nome, url:url, tipo:'gravado'}); salvar(); render(); // 2. SALVA AUTOMÁTICO NA GALERIA DO CELULAR let a=document.createElement('a'); a.href=url; a.download=nome; // Isso faz baixar pra galeria document.body.appendChild(a); a.click(); document.body.removeChild(a); // 3. Tenta compartilhar direto pra galeria (Android novo) if(navigator.canShare && navigator.canShare({files:[new File([blob], nome, {type:'video/mp4'})]})){ navigator.share({files:[new File([blob], nome, {type:'video/mp4'})], title:'MY KNAY', text:nome}); } alert('📼 Vídeo salvo na galeria! Pasta Downloads: '+nome); }; recorder.start(); document.getElementById('gravBtns').innerHTML='🔴 GRAVANDO... clica em PARAR'; showToast('🔴 Gravando...'); } function pararGrav(){ if(recorder && recorder.state!=='inactive'){ recorder.stop(); stream.getTracks().forEach(t=>t.stop()); document.getElementById('cam').style.display='none'; document.getElementById('gravBtns').style.display='none'; showToast('✅ Salvo na galeria'); } } function showToast(t){ let d=document.createElement('div'); d.style='position:fixed;bottom:150px;left:50%;transform:translateX(-50%);background:rgba(0,0,0,.8);color:#fff;padding:10px 18px;border-radius:20px;z-index:9999'; d.innerText=t; document.body.appendChild(d); setTimeout(()=>d.remove(),2000); }
